```js
  /* countdown to 9 August 2026, 19:00 Europe/Berlin */
  function updateCountdown(){
    const target = new Date('2026-08-09T19:00:00+02:00').getTime();
    const now = Date.now();
    const diff = Math.max(0, target - now);
    const days = Math.floor(diff / 86400000);
    const hours = Math.floor((diff % 86400000) / 3600000);
    const mins = Math.floor((diff % 3600000) / 60000);
    const dEl = document.getElementById('cd-days');
    const hEl = document.getElementById('cd-hours');
    const mEl = document.getElementById('cd-mins');
    if(dEl){ dEl.textContent = days; hEl.textContent = String(hours).padStart(2,'0'); mEl.textContent = String(mins).padStart(2,'0'); }
  }
  updateCountdown();
  setInterval(updateCountdown, 30000);
```
